{{- /* A repository file the site renders as it is: CHANGELOG.md, LICENSE.md. The site's Markdown lets raw HTML through (hugo.toml: the how-tos' scenes are HTML), and every branch edits the changelog -- so raw HTML in these files stops the build instead of reaching the public pages. The check looks at what Markdown made of the file, not at its source: in the rendered HTML, text and code are escaped (`` is `<version>`), so every `<` left is a tag. Each one has to be a tag Markdown itself writes, with the attributes it writes. A pattern over the source that skipped code spans could be fooled by a stray backtick. {{ partial "no-raw-html.html" (dict "name" "CHANGELOG.md" "text" .Content "page" $.Page) }} */ -}} {{- $html := .page.RenderString (dict "display" "block") .text -}} {{- $plain := `p|ul|ol|li|em|strong|code|pre|blockquote|del|table|thead|tbody|tr|th|td` -}} {{- $allowed := printf `^(?:|<(?:hr|br) ?/?>|||\s]*"(?: title="[^"<>]*")?>||
    |||
    |
    |
    |||)$` $plain -}}
{{- range findRE `<[^>]*>?` $html -}}
  {{- if not (findRE $allowed .) -}}
    {{- errorf "%s holds raw HTML, which would reach the website as it is: %s -- write it as Markdown or in a code span" $.name . -}}
  {{- end -}}
{{- end -}}
